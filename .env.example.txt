# ==========================================
# ⭐ Group 1 — 個股策略參數（PC_<代號> JSON 格式）
# ==========================================
# 每支股票一個 PC_<代號> 變數，JSON 格式。
#
# 通用參數：
#   strategy      策略名稱（bollinger / vwap / ma_cross / breakout / keep_wait）
#                 或用戶自訂策略（g1_strategy_1 / g1_strategy_2 / g2_strategy_1 / g2_strategy_2）
#   alloc         資金配置百分比（相對於 TOTAL_CAPITAL；bollinger/vwap/ma_cross 單次買入額 = alloc×TOTAL_CAPITAL）
#   BUY_AMOUNT_OFFSET   買入足額容差（預設 0.02）：持倉 ≥ 目標×(1-此值) 視為足額不補買；否則 3 交易日內補足，逾時 TG 通知
#   max_entry_price   最高買入價（-1=不限價立即買入；全輪替換股常用）
#   initial_buy_pct   首次買入比例（0~1；1.0=一次買滿）
#   monthly_budget    每月預算上限（0 = 不限制）
#
# 策略特定參數（對應 strategy 函式參數名稱）：
#   bollinger  | window, std_dev, rsi_period
#   vwap       | sigma_mult, rsi_period
#   ma_cross   | fast_period, slow_period, atr_period, atr_threshold
#   breakout   | lookback, atr_period, atr_threshold, buy_shares, sell_shares
#   keep_wait  | initial_shares, add_drop_pct, add_shares, max_additions,
#              | tp_trigger_pct, tp_sell_ratio, cooldown_days
#
# 金字塔加碼（限 bollinger）：
#   pyramid_enabled, pyramid_tier1_shares, pyramid_tier2_shares,
#   pyramid_tier3_shares, pyramid_tier2_drop, pyramid_tier3_drop
#
# 範例（現行全輪替 keep_wait / ma_cross / breakout 用法）：
PC_3017={"strategy":"keep_wait","alloc":26.7,"max_entry_price":-1,"initial_buy_pct":1.0}
PC_2360={"strategy":"ma_cross","alloc":5,"fast_period":10,"slow_period":30,"atr_period":14,"atr_threshold":0.005}
PC_3008={"strategy":"breakout","alloc":12.5,"buy_shares":14,"sell_shares":14,"lookback":20,"atr_period":14,"atr_threshold":0.02}
